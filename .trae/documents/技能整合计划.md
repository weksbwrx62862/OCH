```python
import sys
import os
sys.path.insert(0, '/home/xxh/金融技能/crypto-analysis/scripts')
from main import CryptoAnalyzer

def get_price_history_from_crypto_analysis(symbol, timeframe):
    """
    从 crypto-analysis 获取价格历史数据
    """
    analyzer = CryptoAnalyzer()
    end_date = datetime.now().strftime("%Y-%m-%d")
    start_date = (datetime.now() - timedelta(days=30)).strftime("%Y-%m-%d")
    history = analyzer.get_crypto_history(symbol, start_date, end_date)

    # 转换格式以匹配 technicals.py 期望的格式
    data = history.get('data', [])
    return {
        'open': [d['price'] for d in data],
        'high': [d['price'] * 1.01 for d in data],  # 简化
        'low': [d['price'] * 0.99 for d in data],   # 简化
        'close': [d['price'] for d in data],
        'volume': [1000 for _ in data]
    }
```
