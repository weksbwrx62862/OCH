'use client';
 
import apiClient from '@/lib/api';
import { useApi } from '@/lib/hooks/useApi';
import { STATUS_COLORS, STATUS_LABELS, formatDate } from '@/lib/utils';
import Link from 'next/link';
 
interface Session {
  id: string;
  title: string;
  status: string;
  agent_id?: string;
  total_messages?: number;
  created_at?: string;
  updated_at?: string;
}
 
export default function SessionsPage() {
  const { data: sessionsRes, loading, refetch } = useApi<{ data: Session[]; total: number }>('/sessions');
  const sessions = sessionsRes?.data || [];
 
  const pauseSession = async (e: React.MouseEvent, sessionId: string) => {
    e.preventDefault();
    e.stopPropagation();
    try {
      await apiClient.put(`/sessions/${sessionId}/pause`);
      refetch();
    } catch (err) { console.error(err); }
  };
 
  const resumeSession = async (e: React.MouseEvent, sessionId: string) => {
    e.preventDefault();
    e.stopPropagation();
    try {
      await apiClient.put(`/sessions/${sessionId}/resume`);
      refetch();
    } catch (err) { console.error(err); }
  };
 
  const deleteSession = async (e: React.MouseEvent, sessionId: string, title: string) => {
    e.preventDefault();
    e.stopPropagation();
    if (!confirm(`确定删除会话「${title}」？`)) return;
    try {
      await apiClient.delete(`/sessions/${sessionId}`);
      refetch();
    } catch (err) { console.error(err); }
  };
 
  return (
    <div className="min-h-screen p-6">
      <div className="max-w-7xl mx-auto">
        <div className="flex items-center justify-between mb-8">
          <div>
            <h1 className="text-2xl font-bold gradient-text">会话管理</h1>
            <p className="text-[var(--text-secondary)] mt-1">所有聊天会话 — 搜索/筛选/导出</p>
          </div>
          <Link
            href="/chat"
            className="px-4 py-2 bg-[var(--primary)] hover:bg-[var(--primary-hover)] text-white text-sm rounded-lg transition-colors"
          >
            + 新建对话
          </Link>
        </div>
 
        <div className="grid grid-cols-4 gap-4 mb-6">
          {[
            { label: '总会话', value: sessions.length },
            { label: '活跃', value: sessions.filter((s) => s.status === 'active').length },
            { label: '已暂停', value: sessions.filter((s) => s.status === 'paused').length },
            { label: '总消息', value: sessions.reduce((acc, s) => acc + (s.total_messages || 0), 0) },
          ].map((stat) => (
            <div key={stat.label} className="bg-[var(--surface)] rounded-xl border border-[var(--border)] p-4">
              <p className="text-xs text-[var(--text-tertiary)]">{stat.label}</p>
              <p className="text-xl font-bold mt-1">{stat.value}</p>
            </div>
          ))}
        </div>
 
        {loading ? (
          <div className="space-y-2">
            {[...Array(5)].map((_, i) => (
              <div key={i} className="h-16 bg-[var(--surface)] rounded-lg border border-[var(--border)] animate-pulse" />
            ))}
          </div>
        ) : sessions.length === 0 ? (
          <div className="text-center py-20">
            <p className="text-[var(--text-secondary)] mb-4">暂无会话</p>
            <Link href="/chat" className="px-4 py-2 bg-[var(--primary)]/20 text-[var(--primary)] rounded-lg text-sm hover:bg-[var(--primary)]/30 transition-colors">
              开始第一个对话
            </Link>
          </div>
        ) : (
          <div className="space-y-2">
            {sessions.map((session) => (
              <Link
                key={session.id}
                href={`/chat?sid=${session.id}`}
                className="group flex items-center justify-between bg-[var(--surface)] rounded-lg border border-[var(--border)] p-4 hover:border-[var(--border-hover)] transition-all"
              >
                <div className="flex items-center gap-4 flex-1 min-w-0">
                  <span className={`w-2 h-2 rounded-full flex-shrink-0 ${
                    session.status === 'active' ? 'bg-emerald-400' :
                    session.status === 'paused' ? 'bg-orange-400' :
                    session.status === 'completed' ? 'bg-blue-400' : 'bg-gray-500'
                  }`} />
                  <div className="min-w-0">
                    <p className="font-medium text-sm truncate group-hover:text-[var(--primary)] transition-colors">
                      {session.title || '未命名会话'}
                    </p>
                    <p className="text-xs text-[var(--text-tertiary)] mt-0.5">
                      {formatDate(session.created_at)} · {session.total_messages || 0} 条消息
                    </p>
                  </div>
                </div>
 
                <div className="flex items-center gap-3 flex-shrink-0">
                  <span className={`px-2 py-0.5 text-xs rounded ${STATUS_COLORS[session.status] || ''}`}>
                    {STATUS_LABELS[session.status] || session.status}
                  </span>
                  <div className="flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
                    {session.status === 'active' && (
                      <button onClick={(e) => pauseSession(e, session.id)} className="px-2 py-1 text-xs rounded bg-orange-500/10 text-orange-400 hover:bg-orange-500/20">暂停</button>
                    )}
                    {session.status === 'paused' && (
                      <button onClick={(e) => resumeSession(e, session.id)} className="px-2 py-1 text-xs rounded bg-emerald-500/10 text-emerald-400 hover:bg-emerald-500/20">恢复</button>
                    )}
                    <button onClick={(e) => deleteSession(e, session.id, session.title || '')} className="px-2 py-1 text-xs rounded bg-red-500/10 text-red-400 hover:bg-red-500/20">删除</button>
                  </div>
                  <svg className="w-4 h-4 text-[var(--text-tertiary)] group-hover:text-[var(--primary)] group-hover:translate-x-1 transition-all" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" /></svg>
                </div>
              </Link>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
