'use client';
 
import { useEffect, useState } from 'react';
import Link from 'next/link';
import apiClient from '@/lib/api';
import { formatDate } from '@/lib/utils';
 
interface DashboardStats {
  activeSessions: number;
  totalAgents: number;
  todayCost: string;
  tokensUsed: string;
}
 
interface AuditLog {
  id: string;
  action: string;
  resource_type: string;
  resource_id: string;
  status_code?: number;
  details?: string;
  created_at: string;
}
 
export default function HomePage() {
  const [stats, setStats] = useState<DashboardStats>({
    activeSessions: 0,
    totalAgents: 0,
    todayCost: '-',
    tokensUsed: '-',
  });
  const [activities, setActivities] = useState<AuditLog[]>([]);
  const [loading, setLoading] = useState(true);
 
  useEffect(() => {
    Promise.all([
      apiClient.get<{ data: { id: string; status: string }[]; total: number }>('/sessions').catch(() => ({ data: [], total: 0 })),
      apiClient.get<{ data: { id: string }[]; total: number }>('/agents').catch(() => ({ data: [], total: 0 })),
      apiClient.get<{ data: AuditLog[]; total: number }>('/audit?per_page=5').catch(() => ({ data: [], total: 0 })),
    ]).then(([sessionsRes, agentsRes, auditRes]) => {
      const activeCount = (sessionsRes.data || []).filter((s: { status: string }) => s.status === 'active').length;
      setStats({
        activeSessions: activeCount,
        totalAgents: agentsRes.total || (agentsRes.data || []).length,
        todayCost: '-',
        tokensUsed: '-',
      });
      setActivities(auditRes.data || []);
      setLoading(false);
    });
  }, []);
 
  return (
    <div className="min-h-screen flex flex-col">
      <header className="h-14 border-b border-[#2a2a2a] flex items-center justify-between px-6 bg-[#111111]/80 backdrop-blur-sm sticky top-0 z-50">
        <div className="flex items-center gap-3">
          <Link href="/" className="flex items-center gap-2">
            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-violet-500 to-cyan-500 flex items-center justify-center">
              <span className="text-white font-bold text-sm">OCH</span>
            </div>
            <span className="font-semibold text-lg">OpenClaw-Harness</span>
          </Link>
        </div>
 
        <nav className="flex items-center gap-1">
          <NavLink href="/" label="Dashboard" active />
          <NavLink href="/chat" label="Chat" />
          <NavLink href="/agents" label="Agents" />
          <NavLink href="/sessions" label="Sessions" />
          <NavLink href="/tasks" label="Tasks" />
          <NavLink href="/tools" label="Tools" />
          <NavLink href="/skills" label="Skills" />
          <NavLink href="/swarm" label="Swarm" />
          <NavLink href="/audit" label="Audit" />
          <NavLink href="/settings" label="Settings" />
        </nav>
 
        <div className="flex items-center gap-3">
          <div className="w-8 h-8 rounded-full bg-violet-500/20 border border-violet-500/30 flex items-center justify-center">
            <span className="text-xs font-medium text-violet-300">U</span>
          </div>
        </div>
      </header>
 
      <main className="flex-1 p-6">
        <div className="max-w-7xl mx-auto space-y-6">
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
            <StatCard title="活跃会话" value={loading ? '...' : String(stats.activeSessions)} icon="💬" />
            <StatCard title="智能体总数" value={loading ? '...' : String(stats.totalAgents)} icon="🤖" />
            <StatCard title="今日费用" value={stats.todayCost} icon="💰" />
            <StatCard title="Token 用量" value={stats.tokensUsed} icon="⚡" />
          </div>
 
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
            <div className="lg:col-span-2 rounded-xl border border-[#2a2a2a] bg-[#111111] p-5">
              <h2 className="text-lg font-semibold mb-4">最近活动</h2>
              {activities.length === 0 ? (
                <p className="text-sm text-gray-500 py-8 text-center">暂无活动记录</p>
              ) : (
                <div className="space-y-3">
                  {activities.map((log) => (
                    <ActivityItem
                      key={log.id}
                      emoji={log.status_code && log.status_code < 400 ? '✅' : log.action.includes('deny') ? '⚠️' : '🔧'}
                      title={`${log.action} ${log.resource_type}`}
                      description={log.resource_id || log.details || ''}
                      time={formatDate(log.created_at)}
                    />
                  ))}
                </div>
              )}
            </div>
 
            <div className="rounded-xl border border-[#2a2a2a] bg-[#111111] p-5">
              <h2 className="text-lg font-semibold mb-4">快捷操作</h2>
              <div className="space-y-2">
                <QuickActionButton label="新建对话" icon="➕" href="/chat" />
                <QuickActionButton label="管理智能体" icon="🤖" href="/agents" />
                <QuickActionButton label="查看会话" icon="📊" href="/sessions" />
                <QuickActionButton label="任务管理" icon="📋" href="/tasks" />
              </div>
            </div>
          </div>
        </div>
      </main>
    </div>
  );
}
 
function NavLink({ href, label, active }: { href: string; label: string; active?: boolean }) {
  return (
    <Link
      href={href}
      className={`px-3 py-1.5 rounded-md text-sm font-medium transition-colors ${
        active
          ? 'bg-violet-500/10 text-violet-300'
          : 'text-gray-400 hover:text-gray-200 hover:bg-[#1a1a1a]'
      }`}
    >
      {label}
    </Link>
  );
}
 
function StatCard({ title, value, icon }: { title: string; value: string; icon: string }) {
  return (
    <div className="rounded-xl border border-[#2a2a2a] bg-[#111111] p-5 hover:border-[#333] transition-colors">
      <div className="flex items-start justify-between mb-3">
        <p className="text-sm text-gray-400">{title}</p>
        <span className="text-xl">{icon}</span>
      </div>
      <span className="text-2xl font-bold">{value}</span>
    </div>
  );
}
 
function ActivityItem({ emoji, title, description, time }: { emoji: string; title: string; description: string; time: string }) {
  return (
    <div className="flex items-start gap-3 p-3 rounded-lg hover:bg-[#1a1a1a] transition-colors">
      <span className="text-lg mt-0.5">{emoji}</span>
      <div className="flex-1 min-w-0">
        <p className="text-sm font-medium">{title}</p>
        <p className="text-xs text-gray-400 truncate">{description}</p>
      </div>
      <span className="text-xs text-gray-500 whitespace-nowrap">{time}</span>
    </div>
  );
}
 
function QuickActionButton({ label, icon, href }: { label: string; icon: string; href: string }) {
  return (
    <Link
      href={href}
      className="flex items-center gap-2 px-4 py-2.5 rounded-lg border border-[#2a2a2a] hover:border-[#333] hover:bg-[#1a1a1a] transition-all text-sm"
    >
      <span>{icon}</span>
      <span>{label}</span>
    </Link>
  );
}
